class Solution:
    def maxSumOfSquares(self, num: int, sum: int) -> str:
        if sum > 9 * num:
            return ""
        if sum == 0:
            return "0" if num == 1 else ""
        result = []
        for _ in range(num):
            digit = min(9, sum)
            result.append(str(digit))
            sum -= digit
        return "".join(result)
